### Подзадача 4.1: Обернуть в `transaction`

```dart
Future<LearningWord> recordReview({
  required int wordId,
  required ReviewQuality quality,
  DateTime? now,
}) async {
  assert(quality >= 0 && quality <= 5, 'quality must be 0..5');
  final ts = now ?? DateTime.now();
  return await transaction(() async {
    final existing = await (select(learningWords)
          ..where((l) => l.wordId.equals(wordId)))
        .getSingleOrNull();
    if (existing == null) {
      throw StateError('No LearningWord row for wordId=$wordId');
    }
    final next = Sm2.next(
      easeFactor: existing.easeFactor,
      intervalDays: existing.intervalDays,
      repetitions: existing.repetitions,
      quality: quality,
      now: ts,
    );
    await (update(learningWords)
          ..where((l) => l.id.equals(existing.id)))
        .write(LearningWordsCompanion(
          status: Value(next.status),
          easeFactor: Value(next.easeFactor),
          intervalDays: Value(next.intervalDays),
          repetitions: Value(next.repetitions),
          nextReviewAt: Value(next.nextReviewAt),
          lastReviewAt: Value(ts),
        ));
    return (select(learningWords)..where((l) => l.id.equals(existing.id)))
        .getSingle();
  });
}
```

```dart
// lib/core/database/migrations.dart
import 'package:drift/drift.dart';
import '../database/app_database.dart';

/// Versioned migration applied by [AppDatabase.migration] when
/// `onUpgrade` is called with `from < targetVersion`. Each subclass
/// declares the version it implements and the SQL changes to apply.
///
/// Adding a new migration is 4 steps:
///   1. Create a new subclass of [Migration].
///   2. Implement [apply] with additive changes only.
///   3. Add the new migration to [_migrationChain] in source order.
///   4. Bump [AppDatabase.schemaVersion].
abstract class Migration {
  const Migration();
  int get targetVersion;
  Future<void> apply(GeneratedDatabase db);
}

class V9ToV10 extends Migration {
  @override int get targetVersion => 10;
  @override
  Future<void> apply(GeneratedDatabase db) async {
    await db.addColumn(db.reciters, db.reciters.mp3quranId);
    // ... existing v9->v10 column additions
  }
}

// ... etc.

final List<Migration> _migrationChain = [
  V9ToV10(),
  V10ToV11(),
  // ...
];

Future<void> applyMigrations(
  Migrator m,
  int from,
  int to,
) async {
  for (final migration in _migrationChain) {
    if (from < migration.targetVersion && migration.targetVersion <= to) {
      await migration.apply(m.database);
    }
  }
}
```
